Configuration reference¶

Every option at its default, as shipped on rev-245.2.

# examples/full-config-reference.yaml
# =============================================================================
# goscape — FULL CONFIGURATION REFERENCE
# =============================================================================
# This file documents EVERY configuration key the server understands, each set
# to its built-in default. It is a reference, not a starter config: copy only
# the keys you want to override into your own file. For a minimal working
# preset, see examples/bundled/goscape.yaml.
#
# This file is itself a valid config. Verify it without starting the server:
#   go run ./cmd/goscape --config.file examples/full-config-reference.yaml --config.verify=true
#
# -----------------------------------------------------------------------------
# How configuration is resolved (lowest to highest precedence):
#   1. built-in defaults (the values shown below)
#   2. this config file            (--config.file <path>)
#   3. environment variables       (only when expansion is enabled, see below)
#   4. CLI flags                   (e.g. --world.tcp-listen-port 43594)
#
# Strictness: the loader uses strict YAML decoding. An UNKNOWN key is a fatal
# boot error, not a warning — keep keys exactly as spelled here.
#
# Meta-flags (value flags — the `=true` is required; the bare form errors):
#   --config.file=<path>        config file to load
#   --config.verify=true        validate config and exit (no server start)
#   --config.expand-env=true    expand ${VAR} / $VAR in this file from the
#                               environment before parsing
#
# Env-var expansion: with --config.expand-env=true, any ${VAR} in a value is
# replaced by the environment. Example: dsn: ${GOSCAPE_DB:-data/goscape.db}
#
# NOTE: every value below is the built-in DEFAULT, with one exception — the four
# module `enable:` keys are set to `true` so this file is a coherent "run
# everything" config (the code default for each is `false`). A module runs only
# when BOTH its `enable:` is true AND it is included by the active `target`.
# =============================================================================


# ----------------------------------------------------------------------------
# Global settings (top level)
# ----------------------------------------------------------------------------

# Which module(s) to run. The selected target pulls in its dependencies.
#   ondemand  HTTP OnDemand server only
#   world     TCP game server only
#   login     gRPC login service only
#   friends   friends server only
#   all       run everything (default)
# CLI: --target
target: all

# Minimum severity to log. One of: trace | debug | info | warn | error.
# CLI: --log.level   (default: info)
log_level: info

# Log output format. One of: text | json.
# CLI: --log.format   (default: text)
log_format: text

# How the `source` attribute is rendered. One of:
#   relative  module-root-relative path, e.g. modules/world/npc_masks.go:42
#             (stays clickable from the repo root in an IDE) — default
#   short     filename only, e.g. npc_masks.go:42
#   full      the compiler's path: absolute under a plain `go run`, or
#             import-path-rooted under -trimpath builds
# CLI: --log.source   (default: relative)
log_source: relative


# ----------------------------------------------------------------------------
# ondemand — HTTP OnDemand server (serves the client cache + /rs2.cgi bootstrap,
#            optional WebSocket -> world bridge)
# ----------------------------------------------------------------------------
ondemand:
  # Run the ondemand module. Code default: false. (yaml-only; no CLI flag.)
  enable: true

  # --- HTTP listener (inlined dskit server.Config) ---
  # CLI: --ondemand.http-listen-network (default: tcp)
  http_listen_network: tcp
  # CLI: --ondemand.http-listen-address (default: 127.0.0.1)
  http_listen_address: 127.0.0.1
  # CLI: --ondemand.http-listen-port (default: 8080)
  http_listen_port: 8080

  # --- HTTP server timeouts (inlined dskit server.Config) ---
  # CLI: --ondemand.graceful-shutdown-timeout (default: 30s)
  graceful_shutdown_timeout: 30s
  # CLI: --ondemand.http-read-timeout (default: 30s)
  http_server_read_timeout: 30s
  # CLI: --ondemand.http-write-timeout (default: 30s)
  http_server_write_timeout: 30s
  # CLI: --ondemand.http-idle-timeout (default: 120s)
  http_server_idle_timeout: 120s

  # --- Source-IP logging (inlined dskit server.Config) ---
  # CLI: --ondemand.log-source-ips-enabled (default: true)
  log_source_ips_enabled: true
  # Header to read the client IP from. Default mirrors the TS getIp() primary.
  # Setting a custom header REPLACES (not augments) dskit's built-in
  # Forwarded / X-Real-IP / X-Forwarded-For chain. Blank both header and regex
  # to fall back to that chain.
  # CLI: --ondemand.log-source-ips-header (default: cf-connecting-ip)
  log_source_ips_header: cf-connecting-ip
  # Regex whose first capture group is the IP. Default extracts the first
  # comma-separated address (TS .split(',')[0].trim()).
  # CLI: --ondemand.log-source-ips-regex
  log_source_ips_regex: '^\s*([^,]+?)\s*(?:,|$)'
  # Log all matched source IPs instead of only the first.
  # CLI: --ondemand.log-source-ips-full (default: false)
  log_source_ips_full: false

  # --- Static-file + cache serving ---
  # Directory with main_file_cache.dat + idx0..4; archive-0 files are served
  # over HTTP from here. CLI: --ondemand.cache-path (default: ./data/pack)
  cache_path: ./data/pack
  # Directory served as a static-file fallback after named routes miss.
  # CLI: --ondemand.public-dir (default: ./public)
  public_dir: ./public

  # --- /rs2.cgi bootstrap params (mirror the world module's NODE_* values;
  #     keep these in sync with the world module when running both) ---
  # CLI: --ondemand.node-id (default: 10)
  node_id: 10
  # CLI: --ondemand.node-members (default: true)
  node_members: true
  # World TCP port; rs2.cgi emits portoff = node_port - 43594 to the applet.
  # CLI: --ondemand.node-port (default: 43594)
  node_port: 43594
  # Whether rs2.cgi may serve the Java applet template when plugin=1.
  # CLI: --ondemand.node-debug (default: true)
  node_debug: true

  # --- WebSocket -> world bridge ---
  websocket:
    # Serve a WS bridge at "/" that forwards binary frames to the world module's
    # TCP connection handler. CLI: --ondemand.websocket-enable (default: true)
    enable: true
    # Explicit Origin allowlist. Empty => allow all (TS default). Non-empty =>
    # the Origin header must exactly match an entry or the upgrade is rejected.
    # (yaml-only; no CLI flag.)
    allowed_origins: []
    # Max size of a single inbound WS message, in bytes.
    # CLI: --ondemand.websocket-max-payload-bytes (default: 2000)
    max_payload_bytes: 2000
    # Mirror of TS NODE_WS_ONDEMAND: permit WS-originated connections to enter
    # the OnDemand (state-2) path. NOTE: not enforced at the proxy layer today
    # (see PORTING.md rev244-b3-ws-ondemand-gate); recorded for parity.
    # CLI: --ondemand.websocket-ws-ondemand (default: false)
    ws_ondemand: false

  # Inject a per-deployment token cookie into the rs2.cgi client page. When
  # true, pub_pem (below) MUST be set and parse as an RSA public key.
  # CLI: --ondemand.websocket-token-protection (default: false)
  ws_token_protection: false
  # RSA public key (PKIX PEM bytes) used to derive the per-deployment token when
  # ws_token_protection is true. yaml-only; omit unless you enable protection.
  #pub_pem: |
  #  -----BEGIN PUBLIC KEY-----
  #  ...
  #  -----END PUBLIC KEY-----

  # --- Advanced dskit server.Config knobs (inlined; no CLI flag; rarely set) ---
  # Per-module log level override (yaml-only; no CLI flag); falls back to the
  # top-level log_level. Valid levels: debug, info, warn, error.
  # (ondemand inherits the dskit HTTP server layer, which does not support trace.)
  #log_level: info
  # HTTP server read-header timeout (default: 0 = no separate header timeout).
  #http_server_read_header_timeout: 0s
  # Log connections closed before any response was written.
  #http_log_closed_connections_without_response_enabled: false
  # Request-header logging / level / exclusions.
  #log_request_headers: false
  #log_request_at_info_level_enabled: false
  #log_request_exclude_headers_list: ""
  # Strip this prefix from incoming request paths before routing.
  #http_path_prefix: ""


# ----------------------------------------------------------------------------
# database — central database shared by login + friends (independent clients)
# ----------------------------------------------------------------------------
# Central database shared by the login and friends services. Both are
# independent clients of it — with the sqlite backend they must share a
# filesystem; postgres lifts that to a network database, letting login and
# friends run on separate hosts against one database.
database:
  # Backend: sqlite | postgres.
  backend: sqlite
  sqlite:
    # SQLite file path (parent directories are created automatically).
    dsn: data/goscape.db
  postgres:
    # PostgreSQL DSN, e.g. postgres://user:pass@host:5432/goscape?sslmode=disable
    # Required when backend=postgres.
    dsn: ""
    # Max open connections per service pool.
    max_open_conns: 8

  tcp_listen_port: 43594

  # Run in production mode. CLI: --world.node-production (default: false)

# ----------------------------------------------------------------------------
# NOT a config section: pkg/telemetry exists in the codebase but is NOT wired
# into the application config. There is no top-level `telemetry:` block — adding
# one fails strict decoding. This is intentional, not a TODO:
#
#   - pkg/telemetry is a dormant emit seam. The engine calls
#     telemetry.Get().Emit*(...) in a few places (login, world, wealth), but the
#     default Emitter is a no-op and nothing in THIS build calls telemetry.Set()
#     or InitProviders() outside tests — so every event lands in the no-op sink.
#   - app.Config only aggregates configs for modules that run here, and there is
#     no telemetry module. No code path reads telemetry.Config to construct a
#     real Emitter, so a `telemetry:` block would configure nothing.
#   - A real Emitter / transport (OTLP, Kafka, …) is supplied out-of-tree; that
#     is where telemetry.Config is consumed. The public core keeps only the
#     interface, the call sites, and the pkg/eventspb schema.
#
# Wiring telemetry.Config in here would surface keys (kafka / otlp / …) that do
# nothing in this build — strictly worse under strict decoding than omitting it.
# ----------------------------------------------------------------------------
